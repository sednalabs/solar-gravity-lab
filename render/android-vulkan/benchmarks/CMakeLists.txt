cmake_minimum_required(VERSION 3.22.1)
project(solarlab_native_benchmark_proof LANGUAGES CXX)
find_package(Vulkan REQUIRED)
find_package(Threads REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
find_program(GLSLC NAMES glslc REQUIRED)
set(NATIVE "${CMAKE_CURRENT_SOURCE_DIR}/../src/main/cpp")
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/shaders")
file(MAKE_DIRECTORY "${GENERATED}")
add_executable(solarlab-native-benchmark HostBenchmarkMain.cpp
    "${NATIVE}/BenchmarkNative.cpp" "${NATIVE}/BenchmarkGpu.cpp"
    "${NATIVE}/BenchmarkOpenCl.cpp" "${NATIVE}/BenchmarkOverlap.cpp" "${NATIVE}/QcomTileMemory.cpp"
    "${NATIVE}/SolarLabVulkanCapabilities.cpp")
target_compile_features(solarlab-native-benchmark PRIVATE cxx_std_20)
target_compile_options(solarlab-native-benchmark PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
target_include_directories(solarlab-native-benchmark PRIVATE "${GENERATED}")
target_link_libraries(solarlab-native-benchmark PRIVATE Vulkan::Vulkan Threads::Threads ${CMAKE_DL_LIBS})
function(benchmark_shader NAME SOURCE)
    set(SPIRV "${GENERATED}/${NAME}.spv")
    set(HEADER "${GENERATED}/${NAME}.h")
    add_custom_command(OUTPUT "${HEADER}"
        COMMAND "${GLSLC}" -O --target-env=vulkan1.1 ${ARGN} "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SOURCE}" -o "${SPIRV}"
        COMMAND "${CMAKE_COMMAND}" "-DINPUT=${SPIRV}" "-DOUTPUT=${HEADER}" "-DSYMBOL=${NAME}" -P "${NATIVE}/EmbedBenchmarkSpirv.cmake"
        DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SOURCE}" "${NATIVE}/EmbedBenchmarkSpirv.cmake" VERBATIM)
    target_sources(solarlab-native-benchmark PRIVATE "${HEADER}")
endfunction()
benchmark_shader(bench_passive_fp32 passive.comp)
benchmark_shader(bench_passive_fp64 passive.comp -DUSE_FP64=1)
benchmark_shader(bench_visual_fp32 visual.comp)
benchmark_shader(bench_visual_fp16 visual.comp -DUSE_FP16=1)
benchmark_shader(bench_compact_atomic compact.comp)
benchmark_shader(bench_compact_subgroup compact.comp -DUSE_SUBGROUP=1)
benchmark_shader(bench_compact_bda compact_bda.comp)
benchmark_shader(bench_dispatch_args dispatch_args.comp)
benchmark_shader(bench_descriptor_control descriptor.comp)
benchmark_shader(bench_descriptor_indexed descriptor.comp -DUSE_NONUNIFORM=1)
set(SOLARLAB_QCOM_SHADER_DIR "" CACHE PATH "Pinned glslang QCOM shader directory")
if(SOLARLAB_QCOM_SHADER_DIR)
    foreach(NAME invert_control tile_invert box_control box_filter)
        set(INPUT "${SOLARLAB_QCOM_SHADER_DIR}/${NAME}.comp.spv")
        set(OUTPUT "${GENERATED}/qcom_${NAME}.h")
        if(NOT EXISTS "${INPUT}")
            message(FATAL_ERROR "Missing pinned QCOM shader: ${INPUT}")
        endif()
        add_custom_command(OUTPUT "${OUTPUT}"
            COMMAND "${CMAKE_COMMAND}" "-DINPUT=${INPUT}" "-DOUTPUT=${OUTPUT}" "-DSYMBOL=qcom_${NAME}" -P "${NATIVE}/EmbedBenchmarkSpirv.cmake"
            DEPENDS "${INPUT}" "${NATIVE}/EmbedBenchmarkSpirv.cmake" VERBATIM)
        target_sources(solarlab-native-benchmark PRIVATE "${OUTPUT}")
    endforeach()
    target_sources(solarlab-native-benchmark PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../probes/QcomOffscreenProbe.cpp")
    target_compile_definitions(solarlab-native-benchmark PRIVATE SOLARLAB_EMBED_QCOM_SHADERS=1 SOLARLAB_QCOM_PROBE_LIBRARY=1)
endif()
